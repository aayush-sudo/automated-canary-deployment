Reason: confirm the query returns numeric data and establish the unit before comparing it with max: 500.

$queries = @{

  primary = 'histogram_quantile(0.99,sum(rate(istio_request_duration_milliseconds_bucket{reporter="source",destination_workload_namespace="default",destination_workload="canary-demo-flagger-primary"}[2m])) by (le))'
  canary = 'histogram_quantile(0.99,sum(rate(istio_request_duration_milliseconds_bucket{reporter="source",destination_workload_namespace="default",destination_workload="canary-demo-flagger-canary"}[2m])) by (le))'
}
foreach ($name in $queries.Keys) {
